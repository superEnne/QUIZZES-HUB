OOP OUTPUT TRACING QUIZ
Topics: Constructors, this(), Overloading, Static vs Instance, 
        Getters/Setters, Object Independence
Focus: LOGIC of the code, not Java syntax
Format: 75 items — 25 Easy / 25 Intermediate / 25 Hard
Instructions: For each item, WRITE THE EXACT SCREEN OUTPUT produced
              when the given main() code runs. Assume all classes
              compile correctly.

// ---- Class 1: Employee ----
public class Employee {
    private int id;
    private double salary;

    public Employee() {
        this(100, 500);
    }
    public Employee(int id, double salary) {
        this.id = id;
        this.salary = salary;
    }
    public void setId(int idNum) { id = idNum; }
    public int getId() { return id; }
    public void setSalary(double salary) { this.salary = salary; }
    public double getSalary() { return salary; }
    public static void message() {
        System.out.println("Employee created");
    }
}

// ---- Class 2: Calculator (this(...) chaining) ----
public class Calculator {
    private int numA;
    private int numB;
    private double ans;

    public Calculator() {
        this(4, 2);
    }
    public Calculator(int numA) {
        this(6.0, 3.0);
    }
    public Calculator(int numA, int numB) {
        this.ans = numA - numB;
    }
    public Calculator(double numA, double numB) {
        this.ans = numA + numB;
    }
    public double getAns() {
        return ans;
    }
    public double getAns(int n) {
        if (n == 0)
            return ans + ans;
        else
            return ans * ans;
    }
}

// ---- Class 3: MDAS ----
public class MDAS {
    private int numA;
    private int numB;

    public MDAS() {
        numA = 8;
        numB = 4;
    }
    public void setCalculate(int numA, int numB) {
        this.numA = numA;
        this.numB = numB;
    }
    public int getCalculate(int n) {
        if (n == 0) return numA + numB;
        else if (n == 1) return numA - numB;
        else if (n == 2) return numA * numB;
        else return numA / numB;
    }
}

// ---- Class 4: Counter ----
public class Counter {
    private int number = 0;
    private int reused = 0;

    public void add() { number = number + 1; }
    public void add(int x) { number = number + x; }
    public void initialize() {
        number = 0;
        reused = reused + 1;
    }
    public int getNumber() { return number; }
    public int getReused() { return reused; }
}

// ---- Class 5: BankAccount (this() chaining, 3 levels) ----
public class BankAccount {
    private String owner;
    private double balance;

    public BankAccount() {
        this("Unknown");
    }
    public BankAccount(String owner) {
        this(owner, 100);
    }
    public BankAccount(String owner, double balance) {
        this.owner = owner;
        this.balance = balance;
    }
    public void deposit(double amt) { balance = balance + amt; }
    public void withdraw(double amt) { balance = balance - amt; }
    public double getBalance() { return balance; }
    public String getOwner() { return owner; }
}

// ---- Class 6: Item (static field shared by all objects) ----
public class Item {
    private String name;
    private static int count = 0;

    public Item(String name) {
        this.name = name;
        count = count + 1;
    }
    public String getName() { return name; }
    public int getMyCount() { return count; }
    public static int getCount() { return count; }
}

// ---- Class 7: Rectangle (this() chaining with defaults) ----
public class Rectangle {
    private double length;
    private double width;

    public Rectangle() {
        this(1, 1);
    }
    public Rectangle(double side) {
        this(side, side);
    }
    public Rectangle(double length, double width) {
        this.length = length;
        this.width = width;
    }
    public double area() { return length * width; }
    public double perimeter() { return 2 * (length + width); }
    public void setLength(double length) { this.length = length; }
}

====================================================================
KEY TAKEAWAYS THIS QUIZ IS TESTING
====================================================================
- this(...) chains to another constructor in the SAME class; trace
  which constructor actually runs before assuming a shortcut value.
- Overload resolution is based on the NUMBER and TYPE of arguments,
  not on argument names or "closest guess."
- A static field belongs to the CLASS, not to any one object — every
  object reading it sees the same, most up-to-date value.
- Assigning one object reference to another (b2 = b1) does NOT copy
  the object; both names point to the same object in memory.
- Method calls that change state (setters, add(), deposit(), etc.)
  permanently affect the object until changed again — always update
  your mental "state table" for the object step by step.
